# v2.0 On-Device AI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add on-device AI features (Core ML) to the iOS app — auto-captions, smart cut, background removal, auto-reframe, scene detection, audio enhancement, smart color match, and object tracking — all running locally with no internet required.

**Architecture:** Each AI feature is a standalone service in `Core/AI/` that takes input (video frames, audio buffers) and produces output (subtitles, masks, crop rects, etc.). A shared `ModelManager` handles downloading, caching, and loading Core ML models on demand. Features integrate into the editor via new UI panels under an "AI" tab and context toolbar actions.

**Tech Stack:** Swift 6, Core ML, Vision framework, AVFoundation, Accelerate framework, SwiftUI

```swift
import Foundation
import CoreML

actor ModelManager {
    enum DownloadError: LocalizedError {
        case networkFailure(underlying: Error)
        case decompressionFailed
        case insufficientDevice
        case cancelled

        var errorDescription: String? {
            switch self {
            case .networkFailure(let error):
                return "Download failed: \(error.localizedDescription)"
            case .decompressionFailed:
                return "Failed to decompress model archive."
            case .insufficientDevice:
                return "This AI feature requires a newer device."
            case .cancelled:
                return "Download was cancelled."
            }
        }
    }

    struct DownloadProgress: Sendable {
        let modelID: AIModelID
        let bytesDownloaded: Int64
        let totalBytes: Int64
        var fraction: Double { Double(bytesDownloaded) / Double(max(totalBytes, 1)) }
    }

    private let cacheDirectory: URL
    private var activeDownloads: [AIModelID: Task<URL, Error>] = [:]
    private var loadedModels: [AIModelID: MLModel] = [:]

    init(cacheDirectory: URL? = nil) {
        if let dir = cacheDirectory {
            self.cacheDirectory = dir
        } else {
            let appSupport = FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            self.cacheDirectory = appSupport.appendingPathComponent("AIModels", isDirectory: true)
        }
        try? FileManager.default.createDirectory(at: self.cacheDirectory, withIntermediateDirectories: true)
    }

    func isDownloaded(_ modelID: AIModelID) -> Bool {
        let manifest = AIModelManifest.manifest(for: modelID)
        let modelDir = cacheDirectory.appendingPathComponent(manifest.fileName)
        let versionFile = modelDir.appendingPathComponent(".version")

        guard FileManager.default.fileExists(atPath: versionFile.path) else { return false }
        guard let cached = try? String(contentsOf: versionFile, encoding: .utf8) else { return false }
        return cached.trimmingCharacters(in: .whitespacesAndNewlines) == manifest.version
    }

    func modelPath(for modelID: AIModelID) -> URL? {
        guard isDownloaded(modelID) else { return nil }
        let manifest = AIModelManifest.manifest(for: modelID)
        return cacheDirectory.appendingPathComponent(manifest.fileName)
    }

    func downloadModel(
        _ modelID: AIModelID,
        progressHandler: (@Sendable (DownloadProgress) -> Void)? = nil
    ) async throws -> URL {
        if let path = modelPath(for: modelID) {
            return path
        }

        if let existing = activeDownloads[modelID] {
            return try await existing.value
        }

        let task = Task<URL, Error> {
            let manifest = AIModelManifest.manifest(for: modelID)
            let destination = cacheDirectory.appendingPathComponent(manifest.fileName)

            let (tempURL, response) = try await URLSession.shared.download(from: manifest.downloadURL)

            guard let httpResponse = response as? HTTPURLResponse,
                  (200...299).contains(httpResponse.statusCode) else {
                throw DownloadError.networkFailure(
                    underlying: URLError(.badServerResponse)
                )
            }

            if FileManager.default.fileExists(atPath: destination.path) {
                try FileManager.default.removeItem(at: destination)
            }

            try FileManager.default.createDirectory(at: destination, withIntermediateDirectories: true)
            try await decompressArchive(at: tempURL, to: destination)

            let versionFile = destination.appendingPathComponent(".version")
            try manifest.version.write(to: versionFile, atomically: true, encoding: .utf8)

            try? FileManager.default.removeItem(at: tempURL)

            return destination
        }

        activeDownloads[modelID] = task

        do {
            let result = try await task.value
            activeDownloads[modelID] = nil
            return result
        } catch {
            activeDownloads[modelID] = nil
            throw error
        }
    }

    func loadModel(_ modelID: AIModelID) async throws -> MLModel {
        if let cached = loadedModels[modelID] {
            return cached
        }

        guard let path = modelPath(for: modelID) else {
            let downloadedPath = try await downloadModel(modelID)
            let compiled = try await MLModel.compileModel(at: downloadedPath)
            let model = try MLModel(contentsOf: compiled)
            loadedModels[modelID] = model
            return model
        }

        let compiled = try await MLModel.compileModel(at: path)
        let model = try MLModel(contentsOf: compiled)
        loadedModels[modelID] = model
        return model
    }

    func unloadModel(_ modelID: AIModelID) {
        loadedModels[modelID] = nil
    }

    func unloadAll() {
        loadedModels.removeAll()
    }

    func deleteModel(_ modelID: AIModelID) {
        unloadModel(modelID)
        let manifest = AIModelManifest.manifest(for: modelID)
        let modelDir = cacheDirectory.appendingPathComponent(manifest.fileName)
        try? FileManager.default.removeItem(at: modelDir)
    }

    func totalCacheSize() -> Int64 {
        guard let enumerator = FileManager.default.enumerator(
            at: cacheDirectory,
            includingPropertiesForKeys: [.fileSizeKey],
            options: [.skipsHiddenFiles]
        ) else { return 0 }

        var total: Int64 = 0
        for case let fileURL as URL in enumerator {
            guard let values = try? fileURL.resourceValues(forKeys: [.fileSizeKey]),
                  let size = values.fileSize else { continue }
            total += Int64(size)
        }
        return total
    }

    private func decompressArchive(at source: URL, to destination: URL) async throws {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: "/usr/bin/ditto")
        process.arguments = ["-xk", source.path, destination.path]

        try process.run()
        process.waitUntilExit()

        guard process.terminationStatus == 0 else {
            throw DownloadError.decompressionFailed
        }
    }
}
```

```swift
import AVFoundation
import CoreML
import Foundation

actor CaptionService {
    enum CaptionError: LocalizedError {
        case modelNotAvailable
        case audioExtractionFailed
        case transcriptionFailed(String)
        case invalidSampleRate

        var errorDescription: String? {
            switch self {
            case .modelNotAvailable:
                return "Caption model is not downloaded. Please download it first."
            case .audioExtractionFailed:
                return "Failed to extract audio from the video."
            case .transcriptionFailed(let detail):
                return "Transcription failed: \(detail)"
            case .invalidSampleRate:
                return "Audio must be resampled to 16kHz."
            }
        }
    }

    private let modelManager: ModelManager
    private let targetSampleRate: Int = 16000
    private let chunkDurationSeconds: Double = 30.0

    init(modelManager: ModelManager) {
        self.modelManager = modelManager
    }

    func transcribeAsset(_ asset: AVAsset, language: CaptionLanguage = .auto) async throws -> CaptionResult {
        let audioBuffer = try await extractAudio(from: asset)
        return try await transcribeBuffer(audioBuffer, sampleRate: targetSampleRate, language: language)
    }

    func transcribeBuffer(
        _ samples: [Float],
        sampleRate: Int,
        language: CaptionLanguage = .auto
    ) async throws -> CaptionResult {
        guard sampleRate == targetSampleRate else {
            let resampled = resample(samples, from: sampleRate, to: targetSampleRate)
            return try await transcribeBuffer(resampled, sampleRate: targetSampleRate, language: language)
        }

        let rms = computeRMS(samples)
        if rms < 0.001 {
            return CaptionResult(segments: [], detectedLanguage: nil, duration: Double(samples.count) / Double(sampleRate))
        }

        let duration = Double(samples.count) / Double(sampleRate)
        let chunkSize = Int(chunkDurationSeconds * Double(sampleRate))
        var allSegments: [CaptionSegment] = []
        var offset: Double = 0

        for chunkStart in stride(from: 0, to: samples.count, by: chunkSize) {
            let chunkEnd = min(chunkStart + chunkSize, samples.count)
            let chunk = Array(samples[chunkStart..<chunkEnd])

            let segments = try await transcribeChunk(chunk, offset: offset, language: language)
            allSegments.append(contentsOf: segments)
            offset += Double(chunkEnd - chunkStart) / Double(sampleRate)
        }

        let detectedLang = allSegments.first?.language ?? (language == .auto ? nil : language.rawValue)
        return CaptionResult(segments: allSegments, detectedLanguage: detectedLang, duration: duration)
    }

    private func transcribeChunk(

        for (index, sample) in samples.enumerated() {

            "audio": MLFeatureValue(multiArray: inputArray)
        ])

        let output = try await model.prediction(from: input)

        return parseWhisperOutput(output, offset: offset)

    func extractAudio(from asset: AVAsset) async throws -> [Float] {
        guard let audioTrack = try await asset.loadTracks(withMediaType: .audio).first else {
            throw CaptionError.audioExtractionFailed
        }

        let duration = try await asset.load(.duration)
        let reader = try AVAssetReader(asset: asset)

        let outputSettings: [String: Any] = [
            AVFormatIDKey: kAudioFormatLinearPCM,
            AVLinearPCMBitDepthKey: 32,
            AVLinearPCMIsFloatKey: true,
            AVLinearPCMIsNonInterleaved: false,
            AVSampleRateKey: targetSampleRate,
            AVNumberOfChannelsKey: 1,
        ]

        let output = AVAssetReaderTrackOutput(track: audioTrack, outputSettings: outputSettings)
        reader.add(output)

        guard reader.startReading() else {
            throw CaptionError.audioExtractionFailed
        }

        var samples: [Float] = []
        let expectedCount = Int(CMTimeGetSeconds(duration) * Double(targetSampleRate))
        samples.reserveCapacity(expectedCount)

        while let buffer = output.copyNextSampleBuffer() {
            guard let blockBuffer = CMSampleBufferGetDataBuffer(buffer) else { continue }
            var length = 0
            var dataPointer: UnsafeMutablePointer<Int8>?
            CMBlockBufferGetDataPointer(blockBuffer, atOffset: 0, lengthAtOffsetOut: nil, totalLengthOut: &length, dataPointerOut: &dataPointer)

            guard let data = dataPointer else { continue }
            let floatCount = length / MemoryLayout<Float>.size
            data.withMemoryRebound(to: Float.self, capacity: floatCount) { floatBuffer in
                samples.append(contentsOf: UnsafeBufferPointer(start: floatBuffer, count: floatCount))
            }
        }

        return samples
    }

    private func resample(_ input: [Float], from sourceSR: Int, to targetSR: Int) -> [Float] {
        guard sourceSR != targetSR else { return input }
        let ratio = Double(targetSR) / Double(sourceSR)
        let outputCount = Int(Double(input.count) * ratio)
        var output = [Float](repeating: 0, count: outputCount)

        for i in 0..<outputCount {
            let srcIndex = Double(i) / ratio
            let srcIndexInt = Int(srcIndex)
            let frac = Float(srcIndex - Double(srcIndexInt))

            let s0 = input[min(srcIndexInt, input.count - 1)]
            let s1 = input[min(srcIndexInt + 1, input.count - 1)]
            output[i] = s0 + frac * (s1 - s0)
        }

        return output
    }

    private func computeRMS(_ samples: [Float]) -> Float {
        guard !samples.isEmpty else { return 0 }
        let sumSquares = samples.reduce(Float(0)) { $0 + $1 * $1 }
        return sqrt(sumSquares / Float(samples.count))
    }
}
```

```swift
import Accelerate
import Foundation

actor SmartCutService {
    func analyze(
        samples: [Float],
        sampleRate: Int,
        configuration: SmartCutConfiguration
    ) -> SmartCutResult {
        guard !samples.isEmpty else {
            return SmartCutResult(silenceSegments: [], fillerWords: [], suggestedCuts: [], totalRemovableDuration: 0)
        }

        let silenceSegments = detectSilence(
            samples: samples,
            sampleRate: sampleRate,
            thresholdDB: configuration.silenceThresholdDB,
            minimumDuration: configuration.minimumSilenceDuration
        )

        let suggestedCuts = silenceSegments.map { segment in
            SmartCutPoint(
                id: UUID().uuidString,
                startTime: segment.startTime + configuration.paddingDuration,
                endTime: segment.endTime - configuration.paddingDuration,
                reason: segment.duration > 2.0 ? .longPause : .silence
            )
        }.filter { $0.endTime > $0.startTime }

        let totalRemovable = suggestedCuts.reduce(0.0) { $0 + $1.duration }

        return SmartCutResult(
            silenceSegments: silenceSegments,
            fillerWords: [],
            suggestedCuts: suggestedCuts,
            totalRemovableDuration: totalRemovable
        )
    }

    func analyzeWithFillers(
        samples: [Float],
        sampleRate: Int,
        captions: CaptionResult,
        configuration: SmartCutConfiguration
    ) -> SmartCutResult {
        let baseResult = analyze(samples: samples, sampleRate: sampleRate, configuration: configuration)

        guard configuration.removeFillersEnabled else {
            return SmartCutResult(
                silenceSegments: baseResult.silenceSegments,
                fillerWords: [],
                suggestedCuts: baseResult.suggestedCuts,
                totalRemovableDuration: baseResult.totalRemovableDuration
            )
        }

        var fillerWords: [FillerWord] = []
        var fillerCuts: [SmartCutPoint] = []

        for segment in captions.segments {
            for word in segment.words {
                let normalized = word.text.lowercased().trimmingCharacters(in: .punctuationCharacters)
                if configuration.fillerWords.contains(normalized) {
                    let filler = FillerWord(
                        id: UUID().uuidString,
                        word: normalized,
                        startTime: word.startTime,
                        endTime: word.endTime,
                        confidence: word.confidence
                    )
                    fillerWords.append(filler)

                    fillerCuts.append(SmartCutPoint(
                        id: UUID().uuidString,
                        startTime: word.startTime,
                        endTime: word.endTime,
                        reason: .fillerWord
                    ))
                }
            }
        }

        let allCuts = (baseResult.suggestedCuts + fillerCuts).sorted { $0.startTime < $1.startTime }
        let mergedCuts = mergOverlappingCuts(allCuts)
        let totalRemovable = mergedCuts.reduce(0.0) { $0 + $1.duration }

        return SmartCutResult(
            silenceSegments: baseResult.silenceSegments,
            fillerWords: fillerWords,
            suggestedCuts: mergedCuts,
            totalRemovableDuration: totalRemovable
        )
    }

    private func detectSilence(
        samples: [Float],
        sampleRate: Int,
        thresholdDB: Float,
        minimumDuration: Double
    ) -> [SilenceSegment] {
        let windowSize = sampleRate / 10 // 100ms windows
        let threshold = pow(10, thresholdDB / 20.0)

        var segments: [SilenceSegment] = []
        var silenceStart: Int?

        for windowStart in stride(from: 0, to: samples.count, by: windowSize) {
            let windowEnd = min(windowStart + windowSize, samples.count)
            let window = Array(samples[windowStart..<windowEnd])

            let rms = computeRMS(window)

            if rms < threshold {
                if silenceStart == nil {
                    silenceStart = windowStart
                }
            } else {
                if let start = silenceStart {
                    let startTime = Double(start) / Double(sampleRate)
                    let endTime = Double(windowStart) / Double(sampleRate)
                    let duration = endTime - startTime

                    if duration >= minimumDuration {
                        segments.append(SilenceSegment(
                            id: UUID().uuidString,
                            startTime: startTime,
                            endTime: endTime
                        ))
                    }
                    silenceStart = nil
                }
            }
        }

        if let start = silenceStart {
            let startTime = Double(start) / Double(sampleRate)
            let endTime = Double(samples.count) / Double(sampleRate)
            if endTime - startTime >= minimumDuration {
                segments.append(SilenceSegment(
                    id: UUID().uuidString,
                    startTime: startTime,
                    endTime: endTime
                ))
            }
        }

        return segments
    }

    private func mergOverlappingCuts(_ cuts: [SmartCutPoint]) -> [SmartCutPoint] {
        guard !cuts.isEmpty else { return [] }

        var merged: [SmartCutPoint] = [cuts[0]]

        for cut in cuts.dropFirst() {
            let last = merged[merged.count - 1]
            if cut.startTime <= last.endTime {
                merged[merged.count - 1] = SmartCutPoint(
                    id: last.id,
                    startTime: last.startTime,
                    endTime: max(last.endTime, cut.endTime),
                    reason: last.reason
                )
            } else {
                merged.append(cut)
            }
        }

        return merged
    }

    private func computeRMS(_ samples: [Float]) -> Float {
        guard !samples.isEmpty else { return 0 }
        var sumSquares: Float = 0
        vDSP_measqv(samples, 1, &sumSquares, vDSP_Length(samples.count))
        return sqrt(sumSquares)
    }
}
```

```swift
import CoreImage
import CoreML
import Foundation
import Vision

actor BackgroundRemovalService {
    enum RemovalError: LocalizedError {
        case segmentationFailed
        case invalidPixelBuffer
        case modelNotAvailable

        var errorDescription: String? {
            switch self {
            case .segmentationFailed: return "Failed to segment the image."
            case .invalidPixelBuffer: return "Invalid video frame."
            case .modelNotAvailable: return "Background removal model not downloaded."
            }
        }
    }

    enum Quality: Sendable {
        case fast       // Vision framework built-in (no model download needed)
        case balanced   // Vision framework with refinement
        case accurate   // SAM mobile Core ML model
    }

    private let modelManager: ModelManager
    private let ciContext = CIContext(options: [.useSoftwareRenderer: false])

    init(modelManager: ModelManager) {
        self.modelManager = modelManager
    }

    func generateMatte(
        from pixelBuffer: CVPixelBuffer,
        quality: Quality = .balanced
    ) async throws -> CIImage {
        switch quality {
        case .fast, .balanced:
            return try await visionSegmentation(pixelBuffer: pixelBuffer, quality: quality)
        case .accurate:
            return try await samSegmentation(pixelBuffer: pixelBuffer)
        }
    }

    func applyRemoval(
        frame: CIImage,
        matte: CIImage,
        replacement: CIImage? = nil
    ) -> CIImage {
        let background = replacement ?? CIImage(color: .clear).cropped(to: frame.extent)

        guard let blendFilter = CIFilter(name: "CIBlendWithMask") else {
            return frame
        }

        blendFilter.setValue(frame, forKey: kCIInputImageKey)
        blendFilter.setValue(background, forKey: kCIInputBackgroundImageKey)
        blendFilter.setValue(matte, forKey: kCIInputMaskImageKey)

        return blendFilter.outputImage ?? frame
    }

    func processFrame(
        pixelBuffer: CVPixelBuffer,
        quality: Quality = .balanced,
        replacement: CIImage? = nil
    ) async throws -> CIImage {
        let frame = CIImage(cvPixelBuffer: pixelBuffer)
        let matte = try await generateMatte(from: pixelBuffer, quality: quality)
        return applyRemoval(frame: frame, matte: matte, replacement: replacement)
    }

    private func visionSegmentation(pixelBuffer: CVPixelBuffer, quality: Quality) async throws -> CIImage {
        let request = VNGeneratePersonSegmentationRequest()
        request.qualityLevel = quality == .fast ? .fast : .balanced
        request.outputPixelFormat = kCVPixelFormatType_OneComponent8

        let handler = VNImageRequestHandler(cvPixelBuffer: pixelBuffer, options: [:])
        try handler.perform([request])

        guard let result = request.results?.first,
              let maskBuffer = result.pixelBuffer else {
            throw RemovalError.segmentationFailed
        }

        return CIImage(cvPixelBuffer: maskBuffer)
    }

    private func samSegmentation(pixelBuffer: CVPixelBuffer) async throws -> CIImage {
        guard await modelManager.isDownloaded(.segmentAnything) else {
            return try await visionSegmentation(pixelBuffer: pixelBuffer, quality: .balanced)
        }

        let model = try await modelManager.loadModel(.segmentAnything)

        let width = CVPixelBufferGetWidth(pixelBuffer)
        let height = CVPixelBufferGetHeight(pixelBuffer)

        let inputImage = CIImage(cvPixelBuffer: pixelBuffer)
        let resized = inputImage.transformed(by: CGAffineTransform(
            scaleX: 1024.0 / CGFloat(width),
            y: 1024.0 / CGFloat(height)
        ))

        var resizedBuffer: CVPixelBuffer?
        CVPixelBufferCreate(nil, 1024, 1024, kCVPixelFormatType_32BGRA, nil, &resizedBuffer)

        guard let outputBuffer = resizedBuffer else {
            throw RemovalError.invalidPixelBuffer
        }

        ciContext.render(resized, to: outputBuffer)

        let inputArray = try pixelBufferToMultiArray(outputBuffer)
        let input = try MLDictionaryFeatureProvider(dictionary: [
            "image": MLFeatureValue(multiArray: inputArray)
        ])

        let output = try await model.prediction(from: input)

        guard let maskArray = output.featureValue(for: "mask")?.multiArrayValue else {
            throw RemovalError.segmentationFailed
        }

        let maskImage = multiArrayToImage(maskArray, width: width, height: height)
        return maskImage
    }

    private func pixelBufferToMultiArray(_ buffer: CVPixelBuffer) throws -> MLMultiArray {
        let width = CVPixelBufferGetWidth(buffer)
        let height = CVPixelBufferGetHeight(buffer)

        let array = try MLMultiArray(shape: [1, 3, NSNumber(value: height), NSNumber(value: width)], dataType: .float32)

        CVPixelBufferLockBaseAddress(buffer, .readOnly)
        defer { CVPixelBufferUnlockBaseAddress(buffer, .readOnly) }

        guard let baseAddress = CVPixelBufferGetBaseAddress(buffer) else {
            throw RemovalError.invalidPixelBuffer
        }

        let bytesPerRow = CVPixelBufferGetBytesPerRow(buffer)
        let data = baseAddress.assumingMemoryBound(to: UInt8.self)

        for y in 0..<height {
            for x in 0..<width {
                let offset = y * bytesPerRow + x * 4
                let b = Float(data[offset]) / 255.0
                let g = Float(data[offset + 1]) / 255.0
                let r = Float(data[offset + 2]) / 255.0

                let pixelIndex = y * width + x
                array[pixelIndex] = NSNumber(value: r)
                array[width * height + pixelIndex] = NSNumber(value: g)
                array[2 * width * height + pixelIndex] = NSNumber(value: b)
            }
        }

        return array
    }

    private func multiArrayToImage(_ array: MLMultiArray, width: Int, height: Int) -> CIImage {
        let maskWidth = array.shape[array.shape.count - 1].intValue
        let maskHeight = array.shape[array.shape.count - 2].intValue

        var pixelData = [UInt8](repeating: 0, count: maskWidth * maskHeight)
        for i in 0..<(maskWidth * maskHeight) {
            let value = array[i].floatValue
            pixelData[i] = UInt8(min(max(value * 255, 0), 255))
        }

        let data = Data(pixelData)
        let colorSpace = CGColorSpaceCreateDeviceGray()

        guard let provider = CGDataProvider(data: data as CFData),
              let cgImage = CGImage(
                  width: maskWidth,
                  height: maskHeight,
                  bitsPerComponent: 8,
                  bitsPerPixel: 8,
                  bytesPerRow: maskWidth,
                  space: colorSpace,
                  bitmapInfo: CGBitmapInfo(rawValue: 0),
                  provider: provider,
                  decode: nil,
                  shouldInterpolate: true,
                  intent: .defaultIntent
              ) else {
            return CIImage(color: .white).cropped(to: CGRect(x: 0, y: 0, width: width, height: height))
        }

        let maskCIImage = CIImage(cgImage: cgImage)
        let scaleX = CGFloat(width) / CGFloat(maskWidth)
        let scaleY = CGFloat(height) / CGFloat(maskHeight)
        return maskCIImage.transformed(by: CGAffineTransform(scaleX: scaleX, y: scaleY))
    }
}
```

```swift
import CoreImage
import Foundation

struct SceneBoundary: Sendable, Identifiable, Equatable {
    let id: String
    let time: Double
    let confidence: Float
}

actor SceneDetectionService {
    struct Configuration: Sendable {
        var threshold: Float
        var minimumSceneDuration: Double
        var analysisFrameRate: Double

        static let `default` = Configuration(
            threshold: 0.4,
            minimumSceneDuration: 1.0,
            analysisFrameRate: 5.0
        )
    }

    private let ciContext = CIContext(options: [.useSoftwareRenderer: false])

    func detectSceneBoundaries(
        frames: [(time: Double, image: CIImage)],
        configuration: Configuration = .default
    ) -> [SceneBoundary] {
        guard frames.count > 1 else { return [] }

        var boundaries: [SceneBoundary] = []
        var previousHistogram = computeHistogram(for: frames[0].image)
        var lastCutTime: Double = 0

        for i in 1..<frames.count {
            let currentHistogram = computeHistogram(for: frames[i].image)
            let difference = histogramDifference(previousHistogram, currentHistogram)

            let timeSinceLastCut = frames[i].time - lastCutTime

            if difference > configuration.threshold && timeSinceLastCut >= configuration.minimumSceneDuration {
                boundaries.append(SceneBoundary(
                    id: UUID().uuidString,
                    time: frames[i].time,
                    confidence: min(difference / configuration.threshold, 1.0)
                ))
                lastCutTime = frames[i].time
            }

            previousHistogram = currentHistogram
        }

        return boundaries
    }

    func detectFromAsset(
        url: URL,
        configuration: Configuration = .default
    ) async throws -> [SceneBoundary] {
        let asset = AVAsset(url: url)
        let duration = try await CMTimeGetSeconds(asset.load(.duration))
        let generator = AVAssetImageGenerator(asset: asset)
        generator.appliesPreferredTrackTransform = true
        generator.maximumSize = CGSize(width: 320, height: 180)

        let frameInterval = 1.0 / configuration.analysisFrameRate
        var frames: [(time: Double, image: CIImage)] = []

        var currentTime: Double = 0
        while currentTime < duration {
            let cmTime = CMTime(seconds: currentTime, preferredTimescale: 600)
            if let cgImage = try? await generator.image(at: cmTime).image {
                frames.append((time: currentTime, image: CIImage(cgImage: cgImage)))
            }
            currentTime += frameInterval
        }

        return detectSceneBoundaries(frames: frames, configuration: configuration)
    }

    private func computeHistogram(for image: CIImage) -> [Float] {
        let extent = image.extent
        guard extent.width > 0, extent.height > 0 else {
            return [Float](repeating: 0, count: 768)
        }

        let width = min(Int(extent.width), 64)
        let height = min(Int(extent.height), 64)

        var bitmap = [UInt8](repeating: 0, count: width * height * 4)
        ciContext.render(
            image,
            toBitmap: &bitmap,
            rowBytes: width * 4,
            bounds: CGRect(x: 0, y: 0, width: width, height: height),
            format: .RGBA8,
            colorSpace: CGColorSpaceCreateDeviceRGB()
        )

        var histogram = [Float](repeating: 0, count: 768) // 256 bins * 3 channels

            let offset = i * 4
            let r = Int(bitmap[offset])
            let g = Int(bitmap[offset + 1])
            let b = Int(bitmap[offset + 2])

            histogram[r] += 1
            histogram[256 + g] += 1
            histogram[512 + b] += 1
        }

        let normFactor = Float(max(pixelCount, 1))
        for i in 0..<histogram.count {
            histogram[i] /= normFactor
        }

        return histogram
    }

    private func histogramDifference(_ a: [Float], _ b: [Float]) -> Float {
        guard a.count == b.count else { return 1.0 }
        var sum: Float = 0
        for i in 0..<a.count {
            sum += abs(a[i] - b[i])
        }
        return sum / Float(a.count) * 100
    }
```

```swift
struct ReframeKeyframe: Sendable, Identifiable, Equatable {
    let id: String
    let time: Double
    let cropRect: CGRect
}

struct ReframeResult: Sendable, Equatable {
    let keyframes: [ReframeKeyframe]
    let targetAspectRatio: CGFloat
    let sourceSize: CGSize
}

actor AutoReframeService {
    enum ReframeError: LocalizedError {
        case noSubjectFound
        case invalidAsset
        case trackingFailed

        var errorDescription: String? {
            switch self {
            case .noSubjectFound: return "No subject detected in the video."
            case .invalidAsset: return "Could not read the video file."
            case .trackingFailed: return "Subject tracking failed."
            }
        }
    }

    struct Configuration: Sendable {
        var targetAspectRatio: CGFloat
        var analysisFrameRate: Double
        var smoothingFactor: Double
        var subjectPadding: CGFloat

        static func portrait() -> Configuration {
            Configuration(targetAspectRatio: 9.0 / 16.0, analysisFrameRate: 5.0, smoothingFactor: 0.3, subjectPadding: 0.15)
        }

        static func square() -> Configuration {
            Configuration(targetAspectRatio: 1.0, analysisFrameRate: 5.0, smoothingFactor: 0.3, subjectPadding: 0.1)
        }
    }

    func reframe(
        assetURL: URL,
        configuration: Configuration
    ) async throws -> ReframeResult {
        let asset = AVAsset(url: assetURL)
        let duration = try await CMTimeGetSeconds(asset.load(.duration))
        let tracks = try await asset.loadTracks(withMediaType: .video)

        guard let videoTrack = tracks.first else {
            throw ReframeError.invalidAsset
        }

        let naturalSize = try await videoTrack.load(.naturalSize)
        let generator = AVAssetImageGenerator(asset: asset)
        generator.appliesPreferredTrackTransform = true

        let frameInterval = 1.0 / configuration.analysisFrameRate
        var rawPositions: [(time: Double, rect: CGRect)] = []

        var currentTime: Double = 0
        while currentTime < duration {
            let cmTime = CMTime(seconds: currentTime, preferredTimescale: 600)
            if let cgImage = try? await generator.image(at: cmTime).image {
                if let subjectRect = try? await detectSubject(in: cgImage) {
                    rawPositions.append((time: currentTime, rect: subjectRect))
                }
            }
            currentTime += frameInterval
        }

        guard !rawPositions.isEmpty else {
            throw ReframeError.noSubjectFound
        }

        let smoothed = smoothPositions(rawPositions, factor: configuration.smoothingFactor)

        let keyframes = smoothed.map { position in
            let cropRect = computeCropRect(
                subjectRect: position.rect,
                sourceSize: naturalSize,
                targetAspectRatio: configuration.targetAspectRatio,
                padding: configuration.subjectPadding
            )
            return ReframeKeyframe(
                id: UUID().uuidString,
                time: position.time,
                cropRect: cropRect
            )
        }

        return ReframeResult(
            keyframes: keyframes,
            targetAspectRatio: configuration.targetAspectRatio,
            sourceSize: naturalSize
        )
    }

    private func detectSubject(in image: CGImage) async throws -> CGRect? {
        let request = VNDetectFaceRectanglesRequest()
        let handler = VNImageRequestHandler(cgImage: image, options: [:])
        try handler.perform([request])

        if let face = request.results?.first {
            return face.boundingBox
        }

        let saliencyRequest = VNGenerateAttentionBasedSaliencyImageRequest()
        let saliencyHandler = VNImageRequestHandler(cgImage: image, options: [:])
        try saliencyHandler.perform([saliencyRequest])

        if let saliency = saliencyRequest.results?.first,
           let salientObject = saliency.salientObjects?.first {
            return salientObject.boundingBox
        }

        return CGRect(x: 0.25, y: 0.25, width: 0.5, height: 0.5)
    }

    private func smoothPositions(
        _ positions: [(time: Double, rect: CGRect)],
        factor: Double
    ) -> [(time: Double, rect: CGRect)] {
        guard positions.count > 1 else { return positions }

        var smoothed = positions
        let alpha = CGFloat(factor)

        for i in 1..<smoothed.count {
            let prev = smoothed[i - 1].rect
            let curr = smoothed[i].rect

            let smoothedRect = CGRect(
                x: prev.origin.x * alpha + curr.origin.x * (1 - alpha),
                y: prev.origin.y * alpha + curr.origin.y * (1 - alpha),
                width: prev.width * alpha + curr.width * (1 - alpha),
                height: prev.height * alpha + curr.height * (1 - alpha)
            )

            smoothed[i] = (time: smoothed[i].time, rect: smoothedRect)
        }

        return smoothed
    }

    private func computeCropRect(
        subjectRect: CGRect,
        sourceSize: CGSize,
        targetAspectRatio: CGFloat,
        padding: CGFloat
    ) -> CGRect {
        let subjectCenterX = subjectRect.midX
        let subjectCenterY = subjectRect.midY

        let cropWidth: CGFloat
        let cropHeight: CGFloat

        if targetAspectRatio < 1 {
            cropHeight = 1.0
            cropWidth = cropHeight * targetAspectRatio
        } else {
            cropWidth = 1.0
            cropHeight = cropWidth / targetAspectRatio
        }

        var originX = subjectCenterX - cropWidth / 2
        var originY = subjectCenterY - cropHeight / 2

        originX = max(0, min(originX, 1 - cropWidth))
        originY = max(0, min(originY, 1 - cropHeight))

        return CGRect(x: originX, y: originY, width: cropWidth, height: cropHeight)
    }
}
```

```swift
import Accelerate
import Foundation

actor AudioEnhanceService {
    struct Configuration: Sendable {
        var targetLoudnessDB: Float
        var noiseGateThresholdDB: Float
        var highPassFrequency: Float
        var deEssFrequency: Float
        var deEssThreshold: Float

        static let voice = Configuration(
            targetLoudnessDB: -16,
            noiseGateThresholdDB: -50,
            highPassFrequency: 80,
            deEssFrequency: 6000,
            deEssThreshold: -20
        )

        static let `default` = Configuration(
            targetLoudnessDB: -14,
            noiseGateThresholdDB: -45,
            highPassFrequency: 60,
            deEssFrequency: 7000,
            deEssThreshold: -18
        )
    }

    func enhance(
        samples: [Float],
        sampleRate: Int,
        configuration: Configuration = .default
    ) -> [Float] {
        guard !samples.isEmpty else { return [] }

        var processed = samples
        processed = highPassFilter(processed, sampleRate: sampleRate, cutoff: configuration.highPassFrequency)
        processed = denoise(samples: processed, sampleRate: sampleRate)
        processed = normalize(samples: processed, targetDB: configuration.targetLoudnessDB)
        processed = limiter(processed, ceiling: 0.95)

        return processed
    }

    func normalize(samples: [Float], targetDB: Float = -14) -> [Float] {
        guard !samples.isEmpty else { return samples }

        var peak: Float = 0
        vDSP_maxmgv(samples, 1, &peak, vDSP_Length(samples.count))

        guard peak > 0.0001 else { return samples }

        let targetPeak: Float = pow(10, targetDB / 20.0)
        let gain = min(targetPeak / peak, 10.0)

        var output = [Float](repeating: 0, count: samples.count)
        var gainVar = gain
        vDSP_vsmul(samples, 1, &gainVar, &output, 1, vDSP_Length(samples.count))

        return output
    }

    func denoise(samples: [Float], sampleRate: Int) -> [Float] {
        guard samples.count > 1024 else { return samples }

        let frameSize = 512
        let hopSize = 256
        var output = [Float](repeating: 0, count: samples.count)

        let noiseProfile = estimateNoiseProfile(samples: samples, frameSize: frameSize)

        var position = 0
        while position + frameSize <= samples.count {
            var frame = Array(samples[position..<(position + frameSize)])

            applySpectralSubtraction(&frame, noiseProfile: noiseProfile)

            for i in 0..<frameSize {
                if position + i < output.count {
                    output[position + i] += frame[i]
                }
            }
            position += hopSize
        }

        var normFactor = Float(frameSize) / Float(hopSize)
        var normalized = [Float](repeating: 0, count: output.count)
        vDSP_vsdiv(output, 1, &normFactor, &normalized, 1, vDSP_Length(output.count))

        return normalized
    }

    private func estimateNoiseProfile(samples: [Float], frameSize: Int) -> [Float] {
        let numFrames = min(10, samples.count / frameSize)
        var profile = [Float](repeating: 0, count: frameSize / 2)

        var minEnergy: Float = .infinity
        var quietestFrame = 0

        for i in 0..<numFrames {
            let start = i * frameSize
            let frame = Array(samples[start..<(start + frameSize)])
            var energy: Float = 0
            vDSP_measqv(frame, 1, &energy, vDSP_Length(frameSize))

            if energy < minEnergy {
                minEnergy = energy
                quietestFrame = i
            }
        }

        let start = quietestFrame * frameSize
        let noiseFrame = Array(samples[start..<(start + frameSize)])

        var realParts = [Float](repeating: 0, count: frameSize)
        var imagParts = [Float](repeating: 0, count: frameSize)

        for i in 0..<frameSize {
            realParts[i] = noiseFrame[i]
        }

        for i in 0..<(frameSize / 2) {
            profile[i] = sqrt(realParts[i] * realParts[i] + imagParts[i] * imagParts[i])
        }

        return profile
    }

    private func applySpectralSubtraction(_ frame: inout [Float], noiseProfile: [Float]) {
        let overSubtraction: Float = 2.0
        let floorFactor: Float = 0.01

        for i in 0..<min(frame.count / 2, noiseProfile.count) {
            let magnitude = abs(frame[i])
            let noiseEst = noiseProfile[i] * overSubtraction
            let cleaned = max(magnitude - noiseEst, magnitude * floorFactor)
            frame[i] = frame[i] >= 0 ? cleaned : -cleaned
        }
    }

    private func highPassFilter(_ samples: [Float], sampleRate: Int, cutoff: Float) -> [Float] {
        guard samples.count > 2 else { return samples }

        let rc = 1.0 / (2.0 * Float.pi * cutoff)
        let dt = 1.0 / Float(sampleRate)
        let alpha = rc / (rc + dt)

        var output = [Float](repeating: 0, count: samples.count)
        output[0] = samples[0]

        for i in 1..<samples.count {
            output[i] = alpha * (output[i - 1] + samples[i] - samples[i - 1])
        }

        return output
    }

    private func limiter(_ samples: [Float], ceiling: Float) -> [Float] {
        samples.map { max(-ceiling, min(ceiling, $0)) }
    }
}
```

```swift
import CoreImage
import Foundation

struct ColorMatchResult: Sendable {
    let brightnessAdjustment: Float
    let contrastAdjustment: Float
    let saturationAdjustment: Float
    let temperatureShift: Float
}

actor ColorMatchService {
    private let ciContext = CIContext(options: [.useSoftwareRenderer: false])

    func computeMatch(
        sourceFrame: CIImage,
        referenceFrame: CIImage
    ) -> ColorMatchResult {
        let sourceStats = computeStats(for: sourceFrame)
        let referenceStats = computeStats(for: referenceFrame)

        let brightnessDiff = referenceStats.meanLuminance - sourceStats.meanLuminance
        let contrastRatio = referenceStats.stdLuminance / max(sourceStats.stdLuminance, 0.001)
        let saturationRatio = referenceStats.meanSaturation / max(sourceStats.meanSaturation, 0.001)
        let tempDiff = referenceStats.meanTemperature - sourceStats.meanTemperature

        return ColorMatchResult(
            brightnessAdjustment: brightnessDiff,
            contrastAdjustment: (contrastRatio - 1.0) * 0.5,
            saturationAdjustment: (saturationRatio - 1.0) * 0.5,
            temperatureShift: tempDiff * 0.3
        )
    }

    func applyMatch(to image: CIImage, result: ColorMatchResult) -> CIImage {
        var output = image

        if abs(result.brightnessAdjustment) > 0.01 {
            if let filter = CIFilter(name: "CIColorControls") {
                filter.setValue(output, forKey: kCIInputImageKey)
                filter.setValue(result.brightnessAdjustment, forKey: kCIInputBrightnessKey)
                filter.setValue(1.0 + result.contrastAdjustment, forKey: kCIInputContrastKey)
                filter.setValue(1.0 + result.saturationAdjustment, forKey: kCIInputSaturationKey)
                output = filter.outputImage ?? output
            }
        }

        if abs(result.temperatureShift) > 0.01 {
            if let filter = CIFilter(name: "CITemperatureAndTint") {
                filter.setValue(output, forKey: kCIInputImageKey)
                let neutral = CIVector(x: CGFloat(6500 + result.temperatureShift * 1000), y: 0)
                filter.setValue(neutral, forKey: "inputNeutral")
                output = filter.outputImage ?? output
            }
        }

        return output
    }

    func generateMatchEffect(result: ColorMatchResult) -> OpenReelProject.Effect {
        OpenReelProject.Effect(
            id: UUID().uuidString,
            type: "colorMatch",
            params: [
                "brightness": .number(Double(result.brightnessAdjustment)),
                "contrast": .number(Double(1.0 + result.contrastAdjustment)),
                "saturation": .number(Double(1.0 + result.saturationAdjustment)),
                "temperature": .number(Double(result.temperatureShift)),
            ],
            enabled: true
        )
    }

    private struct FrameStats {
        var meanLuminance: Float
        var stdLuminance: Float
        var meanSaturation: Float
        var meanTemperature: Float
    }

    private func computeStats(for image: CIImage) -> FrameStats {
        let extent = image.extent
        let width = min(Int(extent.width), 64)
        let height = min(Int(extent.height), 64)

        var bitmap = [UInt8](repeating: 0, count: width * height * 4)
        ciContext.render(
            image,
            toBitmap: &bitmap,
            rowBytes: width * 4,
            bounds: CGRect(x: 0, y: 0, width: width, height: height),
            format: .RGBA8,
            colorSpace: CGColorSpaceCreateDeviceRGB()
        )

        var luminances: [Float] = []
        var saturations: [Float] = []
        var temperatures: [Float] = []

        let pixelCount = width * height
        luminances.reserveCapacity(pixelCount)

        for i in 0..<pixelCount {
            let offset = i * 4
            let r = Float(bitmap[offset]) / 255.0
            let g = Float(bitmap[offset + 1]) / 255.0
            let b = Float(bitmap[offset + 2]) / 255.0

            let luminance = 0.299 * r + 0.587 * g + 0.114 * b
            luminances.append(luminance)

            let maxC = max(r, g, b)
            let minC = min(r, g, b)
            let saturation = maxC > 0 ? (maxC - minC) / maxC : 0
            saturations.append(saturation)

            let temperature = (r - b) * 0.5
            temperatures.append(temperature)
        }

        let meanLum = luminances.reduce(0, +) / Float(max(pixelCount, 1))
        let variance = luminances.reduce(0) { $0 + ($1 - meanLum) * ($1 - meanLum) } / Float(max(pixelCount, 1))
        let stdLum = sqrt(variance)
        let meanSat = saturations.reduce(0, +) / Float(max(pixelCount, 1))
        let meanTemp = temperatures.reduce(0, +) / Float(max(pixelCount, 1))

        return FrameStats(
            meanLuminance: meanLum,
            stdLuminance: stdLum,
            meanSaturation: meanSat,
            meanTemperature: meanTemp
        )
    }
}
```

```swift
import AVFoundation
import Foundation
import Vision

struct TrackingPoint: Sendable, Identifiable, Equatable {
    let id: String
    let time: Double
    let position: CGPoint
    let boundingBox: CGRect
    let confidence: Float
}

struct TrackingResult: Sendable, Equatable {
    let points: [TrackingPoint]
    let startTime: Double
    let endTime: Double
}

actor ObjectTrackingService {
    enum TrackingError: LocalizedError {
        case noObjectAtPoint
        case trackingLost
        case invalidAsset

        var errorDescription: String? {
            switch self {
            case .noObjectAtPoint: return "No trackable object found at the selected point."
            case .trackingLost: return "Object tracking was lost."
            case .invalidAsset: return "Could not read the video for tracking."
            }
        }
    }

    func track(
        in assetURL: URL,
        startingAt point: CGPoint,
        initialRect: CGRect,
        fromTime: Double,
        toTime: Double,
        frameRate: Double = 30
    ) async throws -> TrackingResult {
        let asset = AVAsset(url: assetURL)
        let reader = try AVAssetReader(asset: asset)

        let videoTracks = try await asset.loadTracks(withMediaType: .video)
        guard let videoTrack = videoTracks.first else {
            throw TrackingError.invalidAsset
        }

        let outputSettings: [String: Any] = [
            kCVPixelBufferPixelFormatTypeKey as String: kCVPixelFormatType_32BGRA
        ]

        let output = AVAssetReaderTrackOutput(track: videoTrack, outputSettings: outputSettings)
        output.supportsRandomAccess = true
        reader.add(output)

        let startCMTime = CMTime(seconds: fromTime, preferredTimescale: 600)
        let endCMTime = CMTime(seconds: toTime, preferredTimescale: 600)
        reader.timeRange = CMTimeRange(start: startCMTime, end: endCMTime)

        guard reader.startReading() else {
            throw TrackingError.invalidAsset
        }

        let observation = VNDetectedObjectObservation(boundingBox: initialRect)
        let request = VNTrackObjectRequest(detectedObject: observation)
        request.trackingLevel = .accurate

        var points: [TrackingPoint] = []
        let frameInterval = 1.0 / frameRate
        var currentTime = fromTime
        var lastObservation: VNDetectedObjectObservation = observation

        while let sampleBuffer = output.copyNextSampleBuffer() {
            let presentationTime = CMSampleBufferGetPresentationTimeStamp(sampleBuffer)
            let time = CMTimeGetSeconds(presentationTime)

            guard let pixelBuffer = CMSampleBufferGetImageBuffer(sampleBuffer) else { continue }

            let trackRequest = VNTrackObjectRequest(detectedObject: lastObservation)
            trackRequest.trackingLevel = .accurate

            let handler = VNImageRequestHandler(cvPixelBuffer: pixelBuffer, options: [:])
            try handler.perform([trackRequest])

            guard let result = trackRequest.results?.first as? VNDetectedObjectObservation,
                  result.confidence > 0.3 else {
                continue
            }

            lastObservation = result

            let trackingPoint = TrackingPoint(
                id: UUID().uuidString,
                time: time,
                position: CGPoint(x: result.boundingBox.midX, y: result.boundingBox.midY),
                boundingBox: result.boundingBox,
                confidence: result.confidence
            )
            points.append(trackingPoint)

            currentTime += frameInterval
        }

        guard !points.isEmpty else {
            throw TrackingError.trackingLost
        }

        return TrackingResult(
            points: points,
            startTime: fromTime,
            endTime: toTime
        )
    }

    func trackingPointsToKeyframes(
        _ result: TrackingResult,
        clipStartTime: Double,
        sourceSize: CGSize
    ) -> [OpenReelProject.Keyframe] {
        result.points.map { point in
            let normalizedX = point.position.x
            let normalizedY = 1.0 - point.position.y // Flip Y for screen coordinates

            return OpenReelProject.Keyframe(
                id: UUID().uuidString,
                time: point.time - clipStartTime,
                property: "transform.position",
                value: .object([
                    "x": .number(Double(normalizedX) * Double(sourceSize.width)),
                    "y": .number(Double(normalizedY) * Double(sourceSize.height)),
```

```swift
func applySmartCuts(_ cuts: [SmartCutPoint]) {
    guard var project = currentProject,
          let clipID = selectedClipID,
          let trackIndex = project.timeline.tracks.firstIndex(where: { $0.clips.contains(where: { $0.id == clipID }) }),
          let clipIndex = project.timeline.tracks[trackIndex].clips.firstIndex(where: { $0.id == clipID })
    else { return }

    let clip = project.timeline.tracks[trackIndex].clips[clipIndex]
    let sortedCuts = cuts.sorted { $0.startTime > $1.startTime }

    var newClips: [OpenReelProject.Clip] = []
    var remainingClip = clip

    for cut in sortedCuts {
        let relativeStart = cut.startTime - clip.startTime
        let relativeEnd = cut.endTime - clip.startTime

        guard relativeStart >= 0, relativeEnd <= clip.duration else { continue }

        let afterCut = OpenReelProject.Clip(
            id: UUID().uuidString,
            mediaId: remainingClip.mediaId,
            trackId: remainingClip.trackId,
            startTime: cut.endTime,
            duration: remainingClip.startTime + remainingClip.duration - cut.endTime,
            inPoint: remainingClip.inPoint + relativeEnd,
            outPoint: remainingClip.outPoint,
            effects: remainingClip.effects,
            audioEffects: remainingClip.audioEffects,
            transform: remainingClip.transform,
            volume: remainingClip.volume,
            keyframes: [],
            speed: remainingClip.speed,
            reversed: remainingClip.reversed
        )
        newClips.insert(afterCut, at: 0)

        remainingClip = OpenReelProject.Clip(
            id: remainingClip.id,
            mediaId: remainingClip.mediaId,
            trackId: remainingClip.trackId,
            startTime: remainingClip.startTime,
            duration: relativeStart,
            inPoint: remainingClip.inPoint,
            outPoint: remainingClip.inPoint + relativeStart,
            effects: remainingClip.effects,
            audioEffects: remainingClip.audioEffects,
            transform: remainingClip.transform,
            volume: remainingClip.volume,
            keyframes: [],
            speed: remainingClip.speed,
            reversed: remainingClip.reversed
        )
    }

    var updatedClips = project.timeline.tracks[trackIndex].clips
    updatedClips.remove(at: clipIndex)
    updatedClips.insert(contentsOf: [remainingClip] + newClips, at: clipIndex)

    project.timeline.tracks[trackIndex].clips = updatedClips
    project.modifiedAt = Int(Date().timeIntervalSince1970 * 1000)
    updateProject(project)
}
```
